def __init__(self, v_th0,
v_leaky_alpha, v_leaky_beta,
v_leaky_adpt_en,
v_reset, v_init,
v_th_alpha, v_th_beta, v_th_adpt_en,
v_th_incre, v_l, dv,
ref_len, reset_mode) :
super (ExtendedLIF, self). __init__()
self.threshold_accumulate = ThresholdAccumulate(v_th0=v_th0)
self.reset = ResetAfterSpike(
reset_mode=reset_mode, v_reset =v_reset, dv =dv)
self.update = UpdateWithRefractory(v_init=v_init)
self.fire = Fire(surrogate_function=Rectangle)
self.leaky = Leaky(alpha=v_leaky_alpha,
beta=v_leaky_beta, adpt_en =v_leaky_adpt_en)
self.saturate = Saturate(v_l=v_l)
self.refractory = Refractory(ref_len=ref_len)
self.threshold_dynamics = ThresholdDynamics(
v_th_alpha=v_th_alpha, v_th_beta =v_th_beta,
v_th_incre=v_th_incre, v_th_adpt_en =v_th_adpt_en)
def forward(self, u_in, v_th_adpt, v, ref_cnt) :
v_th = self.threshold_accumulate(v_th_adpt)
v_update = self.update(u_in, v, ref_cnt)
spike = self.fire(v_update, v_th)
v = self.reset(v_update, spike)
v = self.leaky(v)
v = self.saturate(v)

ref_cnt = self.refractory(ref_cnt, spike)
return spike, v, v_th_adpt, ref_cnt
