def forward(self, x : torch.Tensor):
if self.restricted:
x = x.clamp(-QModule.activation_absmax, QModule.activation_absmax)
if self.aware_mode:
x = FakeQuantizeFloor.apply (x, QModule.activation_absmax)
out = torch.nn.Conv2d.forward(self, x)
if self.quantization_mode:
out = out.clamp(INT32_MIN, INT32_MAX)
if not(self.is_last_node):
out = (out >> self.bit_shift).clamp(INT8_MIN, INT8_MAX)
